# --- Variables ---
plan_corners = None
origin_set = False
pen_origin = None  # <-- on ajoute cette variable

...

# --- Boucle principale ---
try:
    while True:
        frame = picam2.capture_array()
        frame_bgr = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)

        # 1. Détection des coins bleus
        if not origin_set:
            plan_corners = detect_blue_corners(frame_bgr)
            if plan_corners is not None:
                origin_set = True
                print(f"[Plan] 4 coins détectés.")
        else:
            # Dessiner le cadre
            for i in range(4):
                pt1 = tuple(plan_corners[i])
                pt2 = tuple(plan_corners[(i + 1) % 4])
                cv2.line(frame_bgr, pt1, pt2, (255, 0, 0), 2)

        # 2. Détection du crayon rouge
        pen_pos = detect_red_pen(frame_bgr)
        if pen_pos:
            cv2.circle(frame_bgr, pen_pos, 8, (0, 255, 0), -1)

            if origin_set:
                pos_cm = map_point_to_plan(pen_pos, plan_corners)
                if pos_cm:
                    x_cm, y_cm = pos_cm

                    # Enregistrer l'origine au premier point détecté
                    if pen_origin is None:
                        pen_origin = (x_cm, y_cm)
                        print(f"[Origine crayon] X={pen_origin[0]:.2f} cm, Y={pen_origin[1]:.2f} cm")

                    # Calcul de la position relative par rapport à pen_origin
                    rel_x = x_cm - pen_origin[0]
                    rel_y = y_cm - pen_origin[1]

                    cv2.putText(frame_bgr, f"X={rel_x:.2f}cm, Y={rel_y:.2f}cm", (10, 30),
                                cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)
                    
                    with open(SAVE_FILE, "w") as f:
                        f.write(f"{rel_x:.2f},{rel_y:.2f}")

                    print(f"[Position relative] X={rel_x:.2f} cm, Y={rel_y:.2f} cm")

        # Affichage
        cv2.imshow("Live Stream", frame_bgr)
        key = cv2.waitKey(1) & 0xFF

        if key == ord('q'):
            break
        elif key == ord('r'):
            origin_set = False
            pen_origin = None  # <-- aussi reset l'origine du crayon
            print("[Reset] Nouvelle détection des coins et du point d'origine.")

finally:
    picam2.stop()
    cv2.destroyAllWindows()
    print("✅ Fin de la détection.")
